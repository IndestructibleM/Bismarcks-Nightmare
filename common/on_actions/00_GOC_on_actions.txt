on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = { 
					limit = { 
						coal < 6
					}
					capital_scope = {
						add_resource = {
    					type = coal
    					amount = 24
						}
					}
				}
				else_if = { 
					limit = { 
						num_of_owned_factories > 30
					}
					capital_scope = {
						add_resource = {
    					type = coal
    					amount = 16
						}
					}
				}
			}
			BAY = {
				every_owned_state = {
					add_core_of = BAY
					remove_core_of = GER
					add_claim_by = GER
				}
			}
			BDN = {
				every_owned_state = {
					add_core_of = BAY
					remove_core_of = GER
					add_claim_by = GER
				}
			}
			WUR = {
				every_owned_state = {
					add_core_of = WUR
					remove_core_of = GER
					add_claim_by = GER
				}
			}
		}
	}
}
