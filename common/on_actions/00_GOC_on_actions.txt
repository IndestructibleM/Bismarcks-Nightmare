on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = { 
					limit = { 
						coal < 6
					}
					capital_scope = {
						add_resource = {
    					type = coal
    					amount = 20
						}
					}
				}
				else_if = { 
					limit = { 
						num_of_owned_factories > 30
					}
					capital_scope = {
						add_resource = {
    					type = coal
    					amount = 10
						}
					}
				}
				if = { 
					limit = { 
						grain < 6
					}
					capital_scope = {
						add_resource = {
    					type = grain
    					amount = 24
						}
					}
				}
				if = { 
					limit = { 
						num_of_owned_factories > 30
					}
					capital_scope = {
						add_resource = {
    					type = grain
    					amount = 16
						}
					}
				}
				add_dynamic_modifier = {
					modifier = econ_production_effect
				}
				econ_production_setup = yes
			}
			BAY = {
				every_owned_state = {
					add_core_of = BAY
					remove_core_of = GER
					add_claim_by = GER
				}
			}
			BDN = {
				every_owned_state = {
					add_core_of = BAY
					remove_core_of = GER
					add_claim_by = GER
				}
			}
			WUR = {
				every_owned_state = {
					add_core_of = WUR
					remove_core_of = GER
					add_claim_by = GER
				}
			}
		}
	}
}

on_actions = {
	on_monthly = {
		effect = {
			every_country = {
				econ_production_setup = yes
			}
		}
	}
}
