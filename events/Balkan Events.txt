country_event = {
	id = balkans.8
	title = balkans.8.t
	desc = balkans.8.d

	fire_only_once = yes

	trigger = {
		tag = SER
		BNT = { has_capitulated = yes }
		CRO = { has_capitulated = yes }
		has_war_with = HUN
		has_war_with = BUL
	}

	option = {
		name = balkans.8.a
		annex_country = { target = BNT }
		CRO = { every_owned_state = { transfer_state_to = SER } }
		every_state = {
			limit = { region = 12 }
			transfer_state_to = ROM
		}
		every_state = {
			limit = { is_core_of = GRE is_owned_by = BUL }
			transfer_state_to = GRE
		}
		white_peace = HUN
		white_peace = BUL
	}
}

country_event = {
	id = balkans.10
	title = balkans.10.t
	desc = balkans.10.d

	fire_only_once = yes

	trigger = {
		tag = HUN
		SER = { has_capitulated = yes }
		ROM = { has_capitulated = yes }
		OR = {
			has_war_with = SER
			has_war_with = GRE
		}
	}

	option = {
		name = balkans.10.a
		SER = {
			every_owned_state = {
				limit = { has_core_of = CRO }
				transfer_state_to = CRO
			}
		}
		every_state = {
			limit = { region = 12 }
			transfer_state_to = HUN
		}
		annex_country = { target = BNT }
		767 = { transfer_state_to = BUL }
		793 = { transfer_state_to = BUL }
		every_state = {
			limit = { has_claim_by = BUL is_owned_by = GRE }
			transfer_state_to = BUL
		}
		white_peace = SER
		white_peace = ROM
		white_peace = GRE
	}
}
